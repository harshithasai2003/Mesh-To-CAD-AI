cat > "$NEW_DATA/make_geodata_dataset.py" <<'PY'
import json
import random
import re
import sys
from pathlib import Path

ROOT = Path("/workdir/bt730192/CAD-Coder/freecad_geodata_ft")

IMAGE_DIR = ROOT / "images"
MACRO_DIR = ROOT / "macros"
FCSTD_DIR = ROOT / "fcstd"
GEODATA_DIR = ROOT / "geodata"
OUT_DIR = ROOT / "json"

SEED = 42
TRAIN_PER_SHAPE = 70

OUT_DIR.mkdir(parents=True, exist_ok=True)


def fail(msg):
    print("\nERROR:", msg)
    sys.exit(1)


def get_id(name, pattern):
    m = re.fullmatch(pattern, name, re.I)
    return m.group(1) if m else None


# ------------------------------------------------------------
# Discover files
# ------------------------------------------------------------

images = {}
for p in IMAGE_DIR.iterdir():
    sid = get_id(p.name, r"(\d+_\d{3})\.(png|jpg|jpeg|webp)")
    if sid:
        images[sid] = p

macros = {}
for p in MACRO_DIR.iterdir():
    sid = get_id(p.name, r"(\d+_\d{3})\.py")
    if sid:
        macros[sid] = p

fcstds = {}
for p in FCSTD_DIR.iterdir():
    sid = get_id(p.name, r"(\d+_\d{3})\.FCStd")
    if sid:
        fcstds[sid] = p

geodata = {}
for p in GEODATA_DIR.iterdir():
    sid = get_id(p.name, r"(\d+_\d{3})_geometry\.json")
    if sid:
        geodata[sid] = p


print("Images :", len(images))
print("Macros :", len(macros))
print("FCStd  :", len(fcstds))
print("Geodata:", len(geodata))

if len(images) != 2000:
    fail(f"Expected 2000 images, got {len(images)}")

if len(macros) != 2000:
    fail(f"Expected 2000 macros, got {len(macros)}")

if len(fcstds) != 2000:
    fail(f"Expected 2000 FCStd files, got {len(fcstds)}")

if len(geodata) != 2000:
    fail(f"Expected 2000 geodata files, got {len(geodata)}")


all_ids = set(images)

if set(macros) != all_ids:
    fail("Image/macro IDs do not match")

if set(fcstds) != all_ids:
    fail("Image/FCStd IDs do not match")

if set(geodata) != all_ids:
    fail("Image/geodata IDs do not match")


# ------------------------------------------------------------
# Validate 20 shapes x 100 variants
# ------------------------------------------------------------

groups = {}

for sid in sorted(all_ids):
    base, variant = sid.split("_")
    groups.setdefault(base, []).append(int(variant))

if len(groups) != 20:
    fail(f"Expected 20 shapes, got {len(groups)}")

for base, variants in sorted(groups.items(), key=lambda x: int(x[0])):
    if sorted(variants) != list(range(1, 101)):
        fail(f"Shape {base} does not contain variants 001-100")


print("\nDataset structure OK")
print("20 shapes x 100 variants = 2000")


# ------------------------------------------------------------
# Build records
# ------------------------------------------------------------

SYSTEM_PROMPT = """Generate a FreeCAD Python macro for this CAD model.

Use the provided image AND the provided geodata.
The geodata describes the geometry of the CAD model and must be used
to determine the dimensions, positions, topology and other geometric
properties needed for the macro.

Use the FreeCAD Python API and the Part module.
Do not use CadQuery.
Return only the FreeCAD Python macro."""


records = {}

for sid in sorted(all_ids):

    # Read geodata
    try:
        geo = json.loads(
            geodata[sid].read_text(encoding="utf-8")
        )
    except Exception as e:
        fail(f"Cannot read geodata for {sid}: {e}")

    if not isinstance(geo, dict):
        fail(f"Geodata for {sid} is not a JSON object")

    # Compact but complete geodata representation
    geo_text = json.dumps(
        geo,
        ensure_ascii=False,
        sort_keys=True,
        separators=(",", ":")
    )

    if not geo_text.strip():
        fail(f"Empty geodata for {sid}")

    # Read target macro
    macro_text = macros[sid].read_text(
        encoding="utf-8"
    ).strip()

    if not macro_text:
        fail(f"Empty macro for {sid}")

    if "cadquery" in macro_text.lower():
        fail(f"CadQuery found in macro {sid}")

    # --------------------------------------------------------
    # CRITICAL:
    # Geodata is placed INSIDE the human prompt.
    # Therefore it is actually tokenized as model input.
    # --------------------------------------------------------

    prompt = (
        "<image>\n"
        + SYSTEM_PROMPT
        + "\n\n"
        + "Geodata:\n"
        + geo_text
    )

    records[sid] = {
        "id": sid,
        "image": images[sid].name,
        "fcstd": fcstds[sid].name,
        "geodata": geodata[sid].name,
        "base_id": sid.split("_")[0],
        "variant": int(sid.split("_")[1]),

        "conversations": [
            {
                "from": "human",
                "value": prompt
            },
            {
                "from": "gpt",
                "value": macro_text
            }
        ]
    }


# ------------------------------------------------------------
# 70/30 split PER SHAPE
# ------------------------------------------------------------

rng = random.Random(SEED)

train = []
test = []

for base in sorted(groups.keys(), key=int):

    ids = [
        f"{base}_{v:03d}"
        for v in range(1, 101)
    ]

    rng.shuffle(ids)

    train_ids = ids[:TRAIN_PER_SHAPE]
    test_ids = ids[TRAIN_PER_SHAPE:]

    if len(train_ids) != 70:
        fail(f"{base}: train != 70")

    if len(test_ids) != 30:
        fail(f"{base}: test != 30")

    if set(train_ids) & set(test_ids):
        fail(f"{base}: train/test overlap")

    train.extend(records[x] for x in train_ids)
    test.extend(records[x] for x in test_ids)


rng.shuffle(train)
rng.shuffle(test)


if len(train) != 1400:
    fail(f"Expected 1400 train samples, got {len(train)}")

if len(test) != 600:
    fail(f"Expected 600 test samples, got {len(test)}")


train_ids = {x["id"] for x in train}
test_ids = {x["id"] for x in test}

if train_ids & test_ids:
    fail("Global train/test overlap")


# ------------------------------------------------------------
# Save
# ------------------------------------------------------------

(OUT_DIR / "train.json").write_text(
    json.dumps(train, ensure_ascii=False, indent=2),
    encoding="utf-8"
)

(OUT_DIR / "test.json").write_text(
    json.dumps(test, ensure_ascii=False, indent=2),
    encoding="utf-8"
)

# Inference file
with (OUT_DIR / "test_eval.jsonl").open(
    "w",
    encoding="utf-8"
) as f:

    for r in test:

        f.write(
            json.dumps(
                {
                    "question_id": r["id"],
                    "image": r["image"],
                    "text": r["conversations"][0]["value"]
                },
                ensure_ascii=False
            )
            + "\n"
        )


report = {
    "seed": SEED,
    "total": 2000,
    "train": 1400,
    "test": 600,
    "shapes": 20,
    "variants_per_shape": 100,
    "train_per_shape": 70,
    "test_per_shape": 30,
    "train_ids": sorted(train_ids),
    "test_ids": sorted(test_ids)
}

(OUT_DIR / "split_report.json").write_text(
    json.dumps(report, indent=2),
    encoding="utf-8"
)


print("\nSUCCESS")
print("Training samples:", len(train))
print("Test samples    :", len(test))
print("Overlap         :", len(train_ids & test_ids))

# Show one actual training example
x = train[0]

print("\n================ SAMPLE INPUT ================")
print(x["conversations"][0]["value"][:3000])

print("\n================ SAMPLE TARGET ===============")
print(x["conversations"][1]["value"][:1500])

print("\nCreated:")
print(OUT_DIR / "train.json")
print(OUT_DIR / "test.json")
print(OUT_DIR / "test_eval.jsonl")
print(OUT_DIR / "split_report.json")
PY
