cat > make_freecad_dataset.py <<'PY'
from pathlib import Path
import json
import random
import re
import sys

ROOT = Path("/workdir/bt730192/CAD-Coder/freecad_dataset")

IMAGE_DIR = ROOT / "images"
MACRO_DIR = ROOT / "macros"
FCSTD_DIR = ROOT / "fcstd"
GEODATA_DIR = ROOT / "geodata"
OUT_DIR = ROOT / "json"

SEED = 42
TRAIN_PER_SHAPE = 70
TEST_PER_SHAPE = 30

OUT_DIR.mkdir(parents=True, exist_ok=True)

def fail(msg):
    print(f"\nERROR: {msg}\n")
    sys.exit(1)

def get_id_from_image(path):
    m = re.fullmatch(r"(\d+)_(\d{3})\.(png|jpg|jpeg|webp)", path.name, re.I)
    if not m:
        return None
    return f"{m.group(1)}_{m.group(2)}"

def get_id_from_macro(path):
    m = re.fullmatch(r"(\d+)_(\d{3})\.py", path.name, re.I)
    if not m:
        return None
    return f"{m.group(1)}_{m.group(2)}"

def get_id_from_fcstd(path):
    m = re.fullmatch(r"(\d+)_(\d{3})\.FCStd", path.name, re.I)
    if not m:
        return None
    return f"{m.group(1)}_{m.group(2)}"

def get_id_from_geo(path):
    m = re.fullmatch(r"(\d+)_(\d{3})_geometry\.json", path.name, re.I)
    if not m:
        return None
    return f"{m.group(1)}_{m.group(2)}"

# ------------------------------------------------------------
# Discover files
# ------------------------------------------------------------

images = {}
for p in IMAGE_DIR.iterdir():
    if p.is_file():
        sid = get_id_from_image(p)
        if sid:
            images[sid] = p

macros = {}
for p in MACRO_DIR.iterdir():
    if p.is_file():
        sid = get_id_from_macro(p)
        if sid:
            macros[sid] = p

fcstds = {}
for p in FCSTD_DIR.iterdir():
    if p.is_file():
        sid = get_id_from_fcstd(p)
        if sid:
            fcstds[sid] = p

geodata = {}
for p in GEODATA_DIR.iterdir():
    if p.is_file():
        sid = get_id_from_geo(p)
        if sid:
            geodata[sid] = p

print(f"Images : {len(images)}")
print(f"Macros : {len(macros)}")
print(f"FCStd  : {len(fcstds)}")
print(f"Geodata: {len(geodata)}")

if len(images) != 2000:
    fail(f"Expected 2000 images, found {len(images)}")

if len(macros) != 2000:
    fail(f"Expected 2000 macros, found {len(macros)}")

if len(fcstds) != 2000:
    fail(f"Expected 2000 FCStd files, found {len(fcstds)}")

if len(geodata) != 2000:
    fail(f"Expected 2000 geometry JSON files, found {len(geodata)}")

all_ids = set(images)

missing_macro = sorted(all_ids - set(macros))
missing_fcstd = sorted(all_ids - set(fcstds))
missing_geo = sorted(all_ids - set(geodata))

if missing_macro:
    fail(f"Missing macros, examples: {missing_macro[:10]}")

if missing_fcstd:
    fail(f"Missing FCStd files, examples: {missing_fcstd[:10]}")

if missing_geo:
    fail(f"Missing geodata, examples: {missing_geo[:10]}")

# ------------------------------------------------------------
# Validate exactly 20 shapes x 100 variants
# ------------------------------------------------------------

groups = {}

for sid in sorted(all_ids):
    base, variant = sid.split("_")
    variant_num = int(variant)

    groups.setdefault(base, []).append(variant_num)

if len(groups) != 20:
    fail(f"Expected 20 shapes, found {len(groups)}")

for base, variants in sorted(groups.items(), key=lambda x: int(x[0])):
    variants = sorted(variants)

    if variants != list(range(1, 101)):
        fail(
            f"Shape {base} does not contain exactly variants 001-100. "
            f"Found: {variants}"
        )

print("\nDataset structure: OK")
print("20 shapes x 100 variants = 2000 samples")

# ------------------------------------------------------------
# Build records
# ------------------------------------------------------------

SYSTEM_PROMPT = """Generate a FreeCAD Python macro for this CAD model using the provided geodata.
Use the FreeCAD Python API and the Part module.
Do not use CadQuery.
Return only the FreeCAD Python macro."""

records = {}

for sid in sorted(all_ids):
    try:
        geo = json.loads(geodata[sid].read_text(encoding="utf-8"))
    except Exception as e:
        fail(f"Could not read geodata {geodata[sid]}: {e}")

    if not isinstance(geo, dict):
        fail(f"Geodata must be a JSON object: {geodata[sid]}")

    geo_text = json.dumps(
        geo,
        ensure_ascii=False,
        sort_keys=True,
        separators=(",", ":")
    )

    macro_text = macros[sid].read_text(encoding="utf-8")

    if "cadquery" in macro_text.lower():
        fail(f"CadQuery detected in macro: {macros[sid]}")

    image_path = images[sid]
    fcstd_path = fcstds[sid]

    prompt = (
        "<image>\n"
        + SYSTEM_PROMPT
        + "\n\n"
        + "Geodata:\n"
        + geo_text
    )

    records[sid] = {
        "id": sid,
        "image": image_path.name,
        "fcstd": fcstd_path.name,
        "geodata": geodata[sid].name,
        "base_id": sid.split("_")[0],
        "variant": int(sid.split("_")[1]),
        "conversations": [
            {
                "from": "human",
                "value": prompt
            },
            {
                "from": "gpt",
                "value": macro_text
            }
        ]
    }

# ------------------------------------------------------------
# Random 70/30 split PER SHAPE
# ------------------------------------------------------------

rng = random.Random(SEED)

train = []
test = []

for base in sorted(groups.keys(), key=int):
    ids = [
        f"{base}_{v:03d}"
        for v in range(1, 101)
    ]

    rng.shuffle(ids)

    train_ids = ids[:TRAIN_PER_SHAPE]
    test_ids = ids[TRAIN_PER_SHAPE:]

    if len(train_ids) != 70:
        fail(f"Shape {base}: training count is not 70")

    if len(test_ids) != 30:
        fail(f"Shape {base}: test count is not 30")

    if set(train_ids) & set(test_ids):
        fail(f"Shape {base}: train/test overlap detected")

    train.extend(records[x] for x in train_ids)
    test.extend(records[x] for x in test_ids)

# Final shuffle of the combined sets
rng.shuffle(train)
rng.shuffle(test)

if len(train) != 1400:
    fail(f"Expected 1400 training samples, got {len(train)}")

if len(test) != 600:
    fail(f"Expected 600 test samples, got {len(test)}")

train_ids = {x["id"] for x in train}
test_ids = {x["id"] for x in test}

if train_ids & test_ids:
    fail("Global train/test overlap detected")

# ------------------------------------------------------------
# Write training JSON
# ------------------------------------------------------------

(OUT_DIR / "train.json").write_text(
    json.dumps(train, ensure_ascii=False, indent=2),
    encoding="utf-8"
)

# This contains ground truth and is useful for later evaluation.
(OUT_DIR / "test.json").write_text(
    json.dumps(test, ensure_ascii=False, indent=2),
    encoding="utf-8"
)

# ------------------------------------------------------------
# Write test JSONL for LLaVA inference
# No macro answer is included.
# ------------------------------------------------------------

with (OUT_DIR / "test_eval.jsonl").open("w", encoding="utf-8") as f:
    for r in test:
        human_prompt = r["conversations"][0]["value"]

        item = {
            "question_id": r["id"],
            "image": r["image"],
            "text": human_prompt
        }

        f.write(json.dumps(item, ensure_ascii=False) + "\n")

# ------------------------------------------------------------
# Split report
# ------------------------------------------------------------

report = {
    "seed": SEED,
    "total": 2000,
    "train": 1400,
    "test": 600,
    "shapes": 20,
    "variants_per_shape": 100,
    "train_per_shape": 70,
    "test_per_shape": 30,
    "train_ids": sorted(train_ids),
    "test_ids": sorted(test_ids)
}

(OUT_DIR / "split_report.json").write_text(
    json.dumps(report, indent=2),
    encoding="utf-8"
)

print("\nSUCCESS")
print("Training samples :", len(train))
print("Test samples     :", len(test))
print("Train/test overlap:", len(train_ids & test_ids))
print("\nCreated:")
print(OUT_DIR / "train.json")
print(OUT_DIR / "test.json")
print(OUT_DIR / "test_eval.jsonl")
print(OUT_DIR / "split_report.json")
PY

#!/bin/bash
set -e

cd /workdir/bt730192/CAD-Coder

PYTHON=/workdir/bt730192/conda/envs/llava/bin/python

export BASE=/workdir/bt730192/cadcoder_data/models/CAD-Coder

export IMAGES=$DATA/images
export OUT=/workdir/bt730192/CAD-Coder/checkpoints/freecad_macro_2000

    echo "Usage: $0 BATCH_ID"
    exit 1
fi

BATCH_FILE="$OUT/test_batches/batch_$(printf '%02d' "$BATCH_ID").jsonl"
ANSWER_FILE="$OUT/test_answers/batch_$(printf '%02d' "$BATCH_ID")_answers.jsonl"
LOG_FILE="$OUT/test_logs/batch_$(printf '%02d' "$BATCH_ID").log"

if [ ! -f "$BATCH_FILE" ]; then
    echo "ERROR: batch file not found: $BATCH_FILE"
    exit 1
fi

if [ -f "$ANSWER_FILE" ]; then
    echo "Already completed: $ANSWER_FILE"
    exit 0
fi

CKPT=$(find "$OUT" -maxdepth 1 -type d -name "checkpoint-*" | sort -V | tail -1)

if [ ! -d "$CKPT" ]; then
    echo "ERROR: checkpoint not found"
    exit 1
fi

echo "Checkpoint: $CKPT"
echo "Batch: $BATCH_FILE"
echo "Output: $ANSWER_FILE"

    --model-path "$CKPT" \
    --question-file "$BATCH_FILE" \
    --image-folder "$IMAGES" \
    --answers-file "$ANSWER_FILE" \
    --temperature 0 \
    --conv-mode llava_v1 \
    --max_new_tokens 2048 \
    2>&1 | tee "$LOG_FILE"

echo "DONE: $ANSWER_FILE"
SH
